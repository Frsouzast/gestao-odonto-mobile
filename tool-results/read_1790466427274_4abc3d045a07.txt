   274→export function ProcedimentosTab() {
   275→  const podeEditar = useAuth((s) => s.podeEditar());
   276→
   277→  const q = useQuery<Procedimento[]>({
   278→    queryKey: ["procedimentos"],
   279→    queryFn: () => apiFetch<Procedimento[]>("/api/procedimentos"),
   280→  });
   281→
   282→  const [selectedId, setSelectedId] = useState<string | null>(null);
   283→
   284→  // Derived effective selection (no setState-in-effect) — auto-selects the
   285→  // first item once data arrives and falls back when the selected id is no
   286→  // longer in the list (e.g., after a delete).
   287→  const effectiveSelectedId = useMemo(() => {
   288→    const list = q.data ?? [];
   289→    if (list.length === 0) return null;
   290→    if (selectedId && list.some((p) => p.id === selectedId)) return selectedId;
   291→    return list[0].id;
   292→  }, [q.data, selectedId]);
   293→
   294→  return (
   295→    <div className="h-full overflow-y-auto scroll-thin">
   296→      <div className="grid grid-cols-1 lg:grid-cols-[260px_1fr] gap-4 p-4 lg:p-6">
   297→        <ProcedimentosListPane
   298→          procedimentos={q.data ?? []}
   299→          loading={q.isLoading}
   300→          error={q.error}
   301→          selectedId={effectiveSelectedId}
   302→          onSelect={setSelectedId}
   303→          onRefetch={() => q.refetch()}
   304→        />
   305→        {effectiveSelectedId ? (
   306→          <ProcedimentoDetalhe key={effectiveSelectedId} procedimentoId={effectiveSelectedId} podeEditar={podeEditar} />
   307→        ) : (
   308→          <EmptyDetailPane hasProcedimentos={(q.data ?? []).length > 0} podeEditar={podeEditar} />
   309→        )}
   310→      </div>
   311→    </div>
   312→  );
   313→}

   368→            type="button"
   369→            size="sm"

   374→          </Button>
   375→        )}
   376→      </div>
   377→
   378→      {loading ? (
   379→        <ListSkeleton />
   380→      ) : error ? (
   381→        <ErrorState message={error.message} onRetry={onRefetch} />
   382→      ) : procedimentos.length === 0 ? (
   383→        <EmptyState
   384→          icon={<Inbox size={22} strokeWidth={1.5} />}
   385→          title="Nenhum procedimento"
   386→          hint={
   387→            podeEditar
   388→              ? "Crie o primeiro procedimento para começar a precificar."
   389→              : "Aguarde o gestor cadastrar procedimentos."
   390→          }
   391→          action={
   392→            podeEditar ? (
   393→              <Button
   394→                type="button"
   395→                size="sm"
   396→                onClick={() => setNovoOpen(true)}
   397→                className="bg-[var(--accent-app)] text-white hover:bg-[var(--accent-app-hover)]"
   398→              >
   399→                <Plus size={14} /> Criar procedimento
   400→              </Button>
   401→            ) : undefined
   402→          }
   403→        />
   404→      ) : (
   405→        <div className="max-h-80 overflow-y-auto scroll-thin py-1">
   406→          {procedimentos.map((p) => {

   410→                key={p.id}
   411→                type="button"
   412→                onClick={() => onSelect(p.id)}
   413→                className={`w-full text-left px-3 py-2 text-sm transition-colors border-l-2 ${
   414→                  active
   415→                    ? "bg-[var(--accent-app-soft-bg)] border-[var(--accent-app)] text-[var(--text-app)]"
   416→                    : "bg-transparent border-transparent hover:bg-[var(--bg-app-alt-strong)]/60 text-[var(--text-app)]"
   417→                }`}
   418→              >
   419→                <div className="flex items-center justify-between gap-2">
   420→                  <span className="truncate font-medium">
   421→                    {p.nome || "(sem nome)"}
   422→                  </span>
   423→                  {p.precoFinal != null && p.precoFinal > 0 && (
   424→                    <span className="text-[11px] font-mono tabular-nums text-[var(--text-app-muted)] shrink-0">
   425→                      {brl(p.precoFinal)}
   426→                    </span>
   427→                  )}
   428→                </div>
   429→                <div className="flex items-center gap-3 text-[11px] text-[var(--text-app-faint)] mt-0.5">
   430→                  <span className="flex items-center gap-1">
   431→                    <Clock size={11} /> {num(p.tempoMinutos)} min
   432→                  </span>
   433→                </div>
   434→              </button>
   435→            );
   436→          })}
   437→        </div>
   438→      )}
   439→
   440→      {/* New procedimento dialog */}
   441→      <Dialog open={novoOpen} onOpenChange={setNovoOpen}>
   442→        <DialogContent className="bg-[var(--surface-app)] border-[var(--border-app)]">
   443→          <DialogHeader>
   444→            <DialogTitle className="text-[var(--text-app)]">
   445→              Novo procedimento
   446→            </DialogTitle>
   447→          </DialogHeader>
   448→          <div className="py-2">
   449→            <Label className="text-[11px] text-[var(--text-app-muted)] mb-1 block">
   450→              Nome do procedimento
   451→            </Label>
   452→            <Input
   453→              value={novoNome}

   460→                  const v = novoNome.trim();
   461→                  if (v) addMut.mutate(v);
   462→                }

   465→            <p className="text-[11px] text-[var(--text-app-muted)] mt-2">
   466→              Os demais parâmetros (tempo, percentuais, insumos) podem ser

   468→            </p>
   469→          </div>
   470→          <DialogFooter>
   471→            <DialogClose asChild>
   472→              <Button
   473→                type="button"
   474→                variant="outline"

   531→// ---------------------------------------------------------------------------
   532→// 3. Detail pane
   533→// ---------------------------------------------------------------------------
   534→function ProcedimentoDetalhe({
   535→  procedimentoId,
   536→  podeEditar,
   537→}: {
   538→  procedimentoId: string;
   539→  podeEditar: boolean;
   540→}) {
   541→  const queryClient = useQueryClient();
   542→
   543→  // Read procedimento from the cached list (shared queryKey, no extra fetch)
   544→  const procQ = useQuery<Procedimento | null>({
   545→    queryKey: ["procedimentos"],
   546→    queryFn: () => apiFetch<Procedimento[]>("/api/procedimentos"),
   547→    select: (list) => list.find((p) => p.id === procedimentoId) ?? null,
   548→  });
   549→
   550→  // Lookup data for the dropdowns
   551→  const insumosQ = useQuery<Insumo[]>({
   552→    queryKey: ["insumos"],
   553→    queryFn: () => apiFetch<Insumo[]>("/api/insumos"),
   554→  });
   555→  const ativosQ = useQuery<Ativo[]>({
   556→    queryKey: ["ativos"],
   557→    queryFn: () => apiFetch<Ativo[]>("/api/ativos"),
   558→  });
   559→
   560→  // Items of this procedimento
   561→  const itensQ = useQuery<ItemProcedimento[]>({
   562→    queryKey: ["procedimentos", procedimentoId, "itens"],
   563→    queryFn: () =>
   564→      apiFetch<ItemProcedimento[]>(
   565→        `/api/procedimentos/${procedimentoId}/itens`
   566→      ),
   567→  });
   568→
   569→  // Calculo (400 = capacidade ausente — exibe banner em vez de erro)
   570→  const calcQ = useQuery<ResultadoCalculo>({
   571→    queryKey: ["procedimentos", procedimentoId, "calculo"],
   572→    queryFn: async () => {
   573→      // Distinguishes 400 (capacidade ausente) from other errors so we can
   574→      // show a friendly banner instead of a hard error.
   575→      const res = await fetch(
   576→        `/api/procedimentos/${procedimentoId}/calculo`,
   577→        {
   578→          headers: {
   579→            Authorization: `Bearer ${useAuth.getState().token ?? ""}`,
   580→          },
   581→        }
   582→      );
   583→      const body = await res.json().catch(() => null);
   584→      if (!res.ok) {
   585→        throw new Error(
   586→          (body && (body.erro || body.error || body.message)) ||
   587→            `Erro ${res.status}`
   588→        );
   589→      }
   590→      return body as ResultadoCalculo;
   591→    },
   592→    retry: false,
   593→  });
   594→
   595→  const proc = procQ.data ?? null;
   596→
   597→  // PUT mutation (single-field inline update on blur)
   598→  const putMut = useMutation({
   599→    mutationFn: (vars: {
   600→      id: string;
   601→      field: string;
   602→      value: number | string | null;
   603→    }) =>

  1469→function InsumoRow({
  1470→  item,
  1471→  insumos,
  1472→  usedInsumoIds,
  1473→  podeEditar,
  1474→  custoUnitario,
  1475→  onTrocarInsumo,
  1476→  onSalvarQuantidade,
  1477→  onRemover,
  1478→  salvando,
  1479→}: {
  1480→  item: ItemProcedimento;
  1481→  insumos: Insumo[];
  1482→  usedInsumoIds: Set<string>;
  1483→  podeEditar: boolean;
  1484→  custoUnitario: number;
  1485→  onTrocarInsumo: (novoInsumoId: string, quantidade: number) => void;
  1486→  onSalvarQuantidade: (quantidade: number) => void;
  1487→  onRemover: () => void;
  1488→  salvando: boolean;

  1490→  const availableInsumos = insumos.filter(
  1491→    (i) => i.id === item.insumoId || !usedInsumoIds.has(i.id)
  1492→  );
  1493→
  1494→  return (
  1495→    <div
  1496→      className={`grid grid-cols-[1fr_80px_110px_40px] gap-2 px-5 py-2 border-b border-[var(--border-app-subtle)] last:border-0 items-center text-sm ${
  1497→        salvando ? "bg-[var(--accent-app-soft-bg)]/40" : ""

  1500→      <select

  1502→        onChange={(e) =>
  1503→          onTrocarInsumo(e.target.value, num(item.quantidade))
  1504→        }
  1505→        disabled={!podeEditar || salvando}
  1506→        className={`w-full h-8 px-2 rounded-md border ${inputCls} text-sm bg-[var(--surface-app)] ${
  1507→          podeEditar ? "" : "opacity-70"

  1512→        )}

  1514→          <option key={i.id} value={i.id}>
  1515→            {i.nome}

  1542→          type="button"
