   274→   274→export function ProcedimentosTab() {
   275→   275→  const podeEditar = useAuth((s) => s.podeEditar());
   276→   276→
   277→   277→  const q = useQuery<Procedimento[]>({
   278→   278→    queryKey: ["procedimentos"],
   279→   279→    queryFn: () => apiFetch<Procedimento[]>("/api/procedimentos"),
   280→   280→  });
   281→   281→
   282→   282→  const [selectedId, setSelectedId] = useState<string | null>(null);
   283→   283→
   284→   284→  // Derived effective selection (no setState-in-effect) — auto-selects the
   285→   285→  // first item once data arrives and falls back when the selected id is no
   286→   286→  // longer in the list (e.g., after a delete).
   287→   287→  const effectiveSelectedId = useMemo(() => {
   288→   288→    const list = q.data ?? [];
   289→   289→    if (list.length === 0) return null;
   290→   290→    if (selectedId && list.some((p) => p.id === selectedId)) return selectedId;
   291→   291→    return list[0].id;
   292→   292→  }, [q.data, selectedId]);
   293→   293→
   294→   294→  return (
   295→   295→    <div className="h-full overflow-y-auto scroll-thin">
   296→   296→      <div className="grid grid-cols-1 lg:grid-cols-[260px_1fr] gap-4 p-4 lg:p-6">
   297→   297→        <ProcedimentosListPane
   298→   298→          procedimentos={q.data ?? []}
   299→   299→          loading={q.isLoading}
   300→   300→          error={q.error}
   301→   301→          selectedId={effectiveSelectedId}
   302→   302→          onSelect={setSelectedId}
   303→   303→          onRefetch={() => q.refetch()}
   304→   304→        />
   305→   305→        {effectiveSelectedId ? (
   306→   306→          <ProcedimentoDetalhe key={effectiveSelectedId} procedimentoId={effectiveSelectedId} podeEditar={podeEditar} />
   307→   307→        ) : (
   308→   308→          <EmptyDetailPane hasProcedimentos={(q.data ?? []).length > 0} podeEditar={podeEditar} />
   309→   309→        )}
   310→   310→      </div>
   311→   311→    </div>
   312→   312→  );
   313→   313→}

   318→   318→function ProcedimentosListPane({
   319→   319→  procedimentos,
   320→   320→  loading,
   321→   321→  error,
   322→   322→  selectedId,
   323→   323→  onSelect,
   324→   324→  onRefetch,
   325→   325→}: {
   326→   326→  procedimentos: Procedimento[];
   327→   327→  loading: boolean;
   328→   328→  error: Error | null;
   329→   329→  selectedId: string | null;
   330→   330→  onSelect: (id: string) => void;
   331→   331→  onRefetch: () => void;
   332→   332→}) {
   333→   333→  const podeEditar = useAuth((s) => s.podeEditar());
   334→   334→  const queryClient = useQueryClient();
   335→   335→  const [novoOpen, setNovoOpen] = useState(false);
   336→   336→  const [novoNome, setNovoNome] = useState("");
   337→   337→
   338→   338→  const addMut = useMutation({
   339→   339→    mutationFn: (nome: string) =>
   340→   340→      apiFetch<Procedimento>("/api/procedimentos", {
   341→   341→        method: "POST",
   342→   342→        body: JSON.stringify({ nome }),
   343→   343→      }),
   344→   344→    onSuccess: (novo) => {
   345→   345→      queryClient.invalidateQueries({ queryKey: ["procedimentos"] });
   346→   346→      setNovoOpen(false);
   347→   347→      setNovoNome("");
   348→   348→      toast.success("Procedimento criado");
   349→   349→      onSelect(novo.id);
   350→   350→    },
   351→   351→    onError: (e: Error) => toast.error(e.message),
   352→   352→  });
   353→   353→
   354→   354→  return (
   355→   355→    <Card className="bg-[var(--surface-app)] border-[var(--border-app)] py-0 shadow-none self-start">
   356→   356→      <div className="px-4 py-3 border-b border-[var(--border-app-subtle)] flex items-center justify-between gap-2">
   357→   357→        <div className="flex items-center gap-2">
   358→   358→          <Stethoscope size={16} className="text-[var(--text-app-muted)]" />
   359→   359→          <h3 className="text-sm font-semibold text-[var(--text-app)]">
   360→   360→            Procedimentos
   361→   361→          </h3>
   362→   362→          <span className="text-[11px] text-[var(--text-app-muted)]">
   363→   363→            {procedimentos.length}
   364→   364→          </span>
   365→   365→        </div>
   366→   366→        {podeEditar && (
   367→   367→          <Button
   368→   368→            type="button"
   369→   369→            size="sm"
   370→   370→            onClick={() => setNovoOpen(true)}
   371→   371→            className="bg-[var(--accent-app)] text-white hover:bg-[var(--accent-app-hover)] h-8"
   372→   372→          >
   373→   373→            <Plus size={14} /> Novo
   374→   374→          </Button>
   375→   375→        )}
   376→   376→      </div>
   377→   377→
   378→   378→      {loading ? (
   379→   379→        <ListSkeleton />
   380→   380→      ) : error ? (
   381→   381→        <ErrorState message={error.message} onRetry={onRefetch} />
   382→   382→      ) : procedimentos.length === 0 ? (
   383→   383→        <EmptyState
   384→   384→          icon={<Inbox size={22} strokeWidth={1.5} />}
   385→   385→          title="Nenhum procedimento"
   386→   386→          hint={
   387→   387→            podeEditar
   388→   388→              ? "Crie o primeiro procedimento para começar a precificar."
   389→   389→              : "Aguarde o gestor cadastrar procedimentos."
   390→   390→          }
   391→   391→          action={
   392→   392→            podeEditar ? (
   393→   393→              <Button
   394→   394→                type="button"
   395→   395→                size="sm"
   396→   396→                onClick={() => setNovoOpen(true)}
   397→   397→                className="bg-[var(--accent-app)] text-white hover:bg-[var(--accent-app-hover)]"
   398→   398→              >
   399→   399→                <Plus size={14} /> Criar procedimento
   400→   400→              </Button>
   401→   401→            ) : undefined
   402→   402→          }
   403→   403→        />
   404→   404→      ) : (
   405→   405→        <div className="max-h-80 overflow-y-auto scroll-thin py-1">
   406→   406→          {procedimentos.map((p) => {
   407→   407→            const active = p.id === selectedId;
   408→   408→            return (
   409→   409→              <button
   410→   410→                key={p.id}
   411→   411→                type="button"
   412→   412→                onClick={() => onSelect(p.id)}
   413→   413→                className={`w-full text-left px-3 py-2 text-sm transition-colors border-l-2 ${
   414→   414→                  active
   415→   415→                    ? "bg-[var(--accent-app-soft-bg)] border-[var(--accent-app)] text-[var(--text-app)]"
   416→   416→                    : "bg-transparent border-transparent hover:bg-[var(--bg-app-alt-strong)]/60 text-[var(--text-app)]"
   417→   417→                }`}
   418→   418→              >
   419→   419→                <div className="flex items-center justify-between gap-2">
   420→   420→                  <span className="truncate font-medium">
   421→   421→                    {p.nome || "(sem nome)"}

   425→   425→                      {brl(p.precoFinal)}
   426→   426→                    </span>
   427→   427→                  )}
   428→   428→                </div>

   431→   431→                    <Clock size={11} /> {num(p.tempoMinutos)} min
   432→   432→                  </span>
   433→   433→                </div>
   434→   434→              </button>
   435→   435→            );
   436→   436→          })}
   437→   437→        </div>
   438→   438→      )}
   439→   439→
   440→   440→      {/* New procedimento dialog */}
   441→   441→      <Dialog open={novoOpen} onOpenChange={setNovoOpen}>
   442→   442→        <DialogContent className="bg-[var(--surface-app)] border-[var(--border-app)]">
   443→   443→          <DialogHeader>
   444→   444→            <DialogTitle className="text-[var(--text-app)]">
   445→   445→              Novo procedimento
   446→   446→            </DialogTitle>
   447→   447→          </DialogHeader>
   448→   448→          <div className="py-2">
   449→   449→            <Label className="text-[11px] text-[var(--text-app-muted)] mb-1 block">
   450→   450→              Nome do procedimento
   451→   451→            </Label>
   452→   452→            <Input

   454→   454→              onChange={(e) => setNovoNome(e.target.value)}
   455→   455→              placeholder="Ex.: Radiografia panorâmica, Periapical, Tratamento de canal…"
   456→   456→              className={inputCls}
   457→   457→              autoFocus
   458→   458→              onKeyDown={(e) => {
   459→   459→                if (e.key === "Enter") {
   460→   460→                  const v = novoNome.trim();
   461→   461→                  if (v) addMut.mutate(v);
   462→   462→                }
   463→   463→              }}
   464→   464→            />
   465→   465→            <p className="text-[11px] text-[var(--text-app-muted)] mt-2">
   466→   466→              Os demais parâmetros (tempo, percentuais, insumos) podem ser

   469→   469→          </div>

   482→   482→              onClick={() => {

   488→   488→                addMut.mutate(v);
   489→   489→              }}

   491→   491→              className="bg-[var(--accent-app)] text-white hover:bg-[var(--accent-app-hover)]"
   492→   492→            >
