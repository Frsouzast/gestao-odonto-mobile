   426→function DespesaRow({
   427→  despesa,
   428→  podeEditar,
   429→  onSalvar,
   430→  onRemover,
   431→  salvando,
   432→}: {
   433→  despesa: Despesa;
   434→  podeEditar: boolean;
   435→  onSalvar: (nome: string, valor: number) => void;
   436→  onRemover: () => void;
   437→  salvando: boolean;
   438→}) {
   439→  // Uncontrolled inputs with defaultValue + key={d.id} (from parent).
   440→  // Initial value comes from the server on first mount; subsequent refetches
   441→  // with the same id keep the user's local edits (no re-init).
   442→  function commitNome(novoNome: string) {
   443→    const n = novoNome.trim();
   444→    if (!n) {
   445→      toast.error("Nome não pode ser vazio");
   446→      return;
   447→    }
   448→    if (n !== despesa.nome) {
   449→      onSalvar(n, num(despesa.valor));
   450→    }
   451→  }
   452→
   453→  function commitValor(novoValor: string) {
   454→    const v = num(novoValor);
   455→    if (!(v >= 0)) {
   456→      toast.error("Valor inválido");
   457→      return;
   458→    }
   459→    if (v !== num(despesa.valor)) {
   460→      onSalvar(despesa.nome, v);
   461→    }
   462→  }
   463→
   464→  return (
   465→    <TableRow className="border-t border-[var(--border-app-subtle)] hover:bg-[var(--bg-app)]/40">
   466→      <TableCell className="px-4 py-1.5">
   467→        <Input
   468→          defaultValue={despesa.nome}
   469→          onBlur={(e) => commitNome(e.currentTarget.value)}
   470→          disabled={!podeEditar || salvando}
   471→          className={`${inputCls} ${cellInputCls} ${
   472→            podeEditar ? "" : "opacity-70"
   473→          }`}
   474→        />
   475→      </TableCell>
   476→      <TableCell className="px-4 py-1.5">
   477→        <Input
   478→          type="number"
   479→          inputMode="decimal"
   480→          step="0.01"
   481→          min="0"
   482→          defaultValue={despesa.valor ?? 0}
   483→          onBlur={(e) => commitValor(e.currentTarget.value)}
   484→          disabled={!podeEditar || salvando}
   485→          className={`${inputCls} ${cellInputCls} text-right font-mono tabular-nums ${
   486→            podeEditar ? "" : "opacity-70"
   487→          }`}
   488→        />
   489→      </TableCell>
   490→      <TableCell className="px-2 text-center">
   491→        <Button
   492→          type="button"
   493→          variant="ghost"
   494→          size="icon"
   495→          className="h-8 w-8 text-[var(--text-app-muted)] hover:text-[var(--danger-app)] hover:bg-[var(--danger-app-bg)]"
   496→          onClick={onRemover}
   497→          disabled={!podeEditar || salvando}
   498→          title="Remover despesa"
   499→        >
   500→          <Trash2 size={14} />
   501→        </Button>
   502→      </TableCell>
   503→    </TableRow>
   504→  );
   505→}

   518→function CapacidadePanel() {
   519→  const capQ = useQuery<Capacidade | null>({
   520→    queryKey: ["capacidade"],
   521→    queryFn: () => apiFetch<Capacidade | null>("/api/capacidade"),
   522→  });
   523→
   524→  // Aux queries to compute custoFixoPorHora preview from despesas+ativos
   525→  const despesasQ = useQuery<Despesa[]>({
   526→    queryKey: ["despesas"],
   527→    queryFn: () => apiFetch<Despesa[]>("/api/despesas"),
   528→  });
   529→  const ativosQ = useQuery<Ativo[]>({
   530→    queryKey: ["ativos"],
   531→    queryFn: () => apiFetch<Ativo[]>("/api/ativos"),
   532→  });
   533→
   534→  const custoFixo = useMemo(() => {
   535→    const desp = (despesasQ.data ?? []).reduce((a, d) => a + num(d.valor), 0);
   536→    const dep = (ativosQ.data ?? []).reduce(
   537→      (a, x) =>
   538→        a +
   539→        (num(x.vidaUtilAnos) > 0
   540→          ? num(x.valorAquisicao) / num(x.vidaUtilAnos) / 12
   541→          : 0),
   542→      0
   543→    );
   544→    return desp + dep;
   545→  }, [despesasQ.data, ativosQ.data]);
   546→
   547→  if (capQ.isLoading) {
   548→    return (
   549→      <Card className="bg-[var(--surface-app)] border-[var(--border-app)] py-0 shadow-none">
   550→        <div className="p-5 space-y-4">
   551→          <Skeleton className="h-5 w-48 bg-[var(--bg-app-alt-strong)]" />
   552→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   553→            {Array.from({ length: 4 }).map((_, i) => (
   554→              <Skeleton
   555→                key={i}
   556→                className="h-16 bg-[var(--bg-app-alt-strong)]"
   557→              />
   558→            ))}
   559→          </div>
   560→        </div>
   561→      </Card>
   562→    );
   563→  }
   564→
   565→  if (capQ.isError) {
   566→    return (
   567→      <ErrorState
   568→        message={capQ.error?.message ?? "Falha ao carregar capacidade"}
   569→        onRetry={() => capQ.refetch()}
   570→      />
   571→    );
   572→  }
   573→
   574→  const semCapacidade = capQ.isSuccess && capQ.data === null;
   575→  // `key` forces remount (and re-init of useState) when transitioning from
   576→  // null → configured capacidade. Stable id while the same data is shown.

   621→    horasEfetivas > 0 && temCustos ? custoFixo / horasEfetivas : null;

   687→                sugeridos — ajuste e salve.
   688→              </div>
   689→            </div>
   690→          </div>
   691→        </Card>
   692→      )}
   693→
   694→      <Card className="bg-[var(--surface-app)] border-[var(--border-app)] py-0 shadow-none">
   695→        <div className="px-4 py-3 border-b border-[var(--border-app-subtle)] flex items-center justify-between gap-2">

   706→            >
   707→              {saveMut.isPending ? "Salvando…" : "Salvar"}
   708→            </Button>

  1678→// ----------------------------------------------------------------------------
  1679→// Resumo fetcher (returns null when capacidade is missing — HTTP 400)
  1680→// ----------------------------------------------------------------------------
  1681→async function fetchResumo(): Promise<Resumo | null> {
  1682→  const token = useAuth.getState().token;
  1683→  const res = await fetch("/api/resumo", {
  1684→    headers: {
  1685→      "Content-Type": "application/json",
  1686→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
  1687→    },
  1688→  });
  1689→  if (res.status === 400) return null; // capacidade not configured — not an error
  1690→  const body = await res.json().catch(() => null);
  1691→  if (!res.ok) {
  1692→    const erro =
  1693→      (body && (body.erro || body.error || body.message)) ||
  1694→      `Erro ${res.status}`;
  1695→    throw new Error(erro);
  1696→  }
  1697→  return body as Resumo;
  1698→}
