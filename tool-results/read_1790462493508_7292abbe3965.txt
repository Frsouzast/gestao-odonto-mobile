   165→   264→// ---- Custo fixo por minuto (agregado) ----
   166→   265→async function carregarCustoMinuto(clinicaId) {
   167→   266→  const [despesasRes, ativosRes, capacidadeRes] = await Promise.all([
   168→   267→    db.query("select valor from despesa_fixa where clinica_id = $1", [clinicaId]),
   169→   268→    db.query("select valor_aquisicao, vida_util_anos from ativo where clinica_id = $1", [clinicaId]),
   170→   269→    db.query("select * from capacidade_produtiva where clinica_id = $1", [clinicaId]),
   171→   270→  ]);
   172→   271→  const despesas = despesasRes.rows.map((d) => ({ valor: Number(d.valor) }));
   173→   272→  // Depreciação mensal de cada ativo = valor de aquisição / vida útil (anos) / 12.
   174→   273→  // Entra na soma do custo fixo do mesmo jeito que uma despesa comum — é
   175→   274→  // dinheiro que a clínica "gasta" todo mês, só que na forma de desgaste do
   176→   275→  // equipamento em vez de uma conta a pagar.
   177→   276→  const depreciacoes = ativosRes.rows.map((a) => ({
   178→   277→    valor: Number(a.vida_util_anos) > 0 ? Number(a.valor_aquisicao) / Number(a.vida_util_anos) / 12 : 0,
   179→   278→  }));
   180→   279→  const depreciacaoMensalTotal = custoFixoTotal(depreciacoes);
   181→   280→  const capacidadeRow = capacidadeRes.rows[0];
   182→   281→  if (!capacidadeRow) return null;
   183→   282→  const capacidade = {
   184→   283→    diasTrabalhados: Number(capacidadeRow.dias_trabalhados),
   185→   284→    horasPorDia: Number(capacidadeRow.horas_por_dia),
   186→   285→    unidadesRenda: Number(capacidadeRow.unidades_renda),
   187→   286→    percentOcupacao: Number(capacidadeRow.percent_ocupacao),
   188→   287→  };
   189→   288→  const todosOsCustos = [...despesas, ...depreciacoes];
   190→   289→  return {
   191→   290→    custoFixoTotal: custoFixoTotal(todosOsCustos),
   192→   291→    depreciacaoMensalTotal,
   193→   292→    horasEfetivas: horasEfetivas(capacidade),
   194→   293→    custoFixoPorHora: custoFixoPorHora(todosOsCustos, capacidade),
   195→   294→    custoFixoPorMinuto: custoFixoPorMinuto(todosOsCustos, capacidade),
   196→   295→  };
   197→   296→}
