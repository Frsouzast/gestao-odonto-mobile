   264→// ---- Custo fixo por minuto (agregado) ----
   265→async function carregarCustoMinuto(clinicaId) {
   266→  const [despesasRes, ativosRes, capacidadeRes] = await Promise.all([
   267→    db.query("select valor from despesa_fixa where clinica_id = $1", [clinicaId]),
   268→    db.query("select valor_aquisicao, vida_util_anos from ativo where clinica_id = $1", [clinicaId]),
   269→    db.query("select * from capacidade_produtiva where clinica_id = $1", [clinicaId]),
   270→  ]);
   271→  const despesas = despesasRes.rows.map((d) => ({ valor: Number(d.valor) }));
   272→  // Depreciação mensal de cada ativo = valor de aquisição / vida útil (anos) / 12.
   273→  // Entra na soma do custo fixo do mesmo jeito que uma despesa comum — é
   274→  // dinheiro que a clínica "gasta" todo mês, só que na forma de desgaste do
   275→  // equipamento em vez de uma conta a pagar.
   276→  const depreciacoes = ativosRes.rows.map((a) => ({
   277→    valor: Number(a.vida_util_anos) > 0 ? Number(a.valor_aquisicao) / Number(a.vida_util_anos) / 12 : 0,
   278→  }));
   279→  const depreciacaoMensalTotal = custoFixoTotal(depreciacoes);
   280→  const capacidadeRow = capacidadeRes.rows[0];
   281→  if (!capacidadeRow) return null;
   282→  const capacidade = {
   283→    diasTrabalhados: Number(capacidadeRow.dias_trabalhados),
   284→    horasPorDia: Number(capacidadeRow.horas_por_dia),
   285→    unidadesRenda: Number(capacidadeRow.unidades_renda),
   286→    percentOcupacao: Number(capacidadeRow.percent_ocupacao),
   287→  };
   288→  const todosOsCustos = [...despesas, ...depreciacoes];
   289→  return {
   290→    custoFixoTotal: custoFixoTotal(todosOsCustos),
   291→    depreciacaoMensalTotal,
   292→    horasEfetivas: horasEfetivas(capacidade),
   293→    custoFixoPorHora: custoFixoPorHora(todosOsCustos, capacidade),
   294→    custoFixoPorMinuto: custoFixoPorMinuto(todosOsCustos, capacidade),
   295→  };
   296→}
