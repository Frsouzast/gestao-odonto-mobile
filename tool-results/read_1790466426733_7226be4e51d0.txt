   159→// ---------- API client ----------
   160→function useApi(token) {
   161→  return useCallback(
   162→    async (path, options = {}) => {
   163→      const res = await fetch(`${API_BASE}/api${path}`, {
   164→        ...options,
   165→        headers: {
   166→          "Content-Type": "application/json",
   167→          Authorization: `Bearer ${token}`,
   168→          ...(options.headers || {}),
   169→        },
   170→      });
   171→      if (!res.ok) {
   172→        let msg = `Erro ${res.status}`;
   173→        try {
   174→          const body = await res.json();
   175→          if (body.erro) msg = body.erro;
   176→        } catch (_) {}
   177→        const erro = new Error(msg);
   178→        erro.status = res.status;
   179→        throw erro;
   180→      }
   181→      if (res.status === 204) return null;
   182→      const texto = await res.text();
   183→      return texto ? JSON.parse(texto) : null; // algumas rotas respondem "criado com sucesso" sem corpo nenhum
   184→    },
   185→    [token]
   186→  );
   187→}
