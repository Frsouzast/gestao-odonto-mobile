   218→   218→function DespesasPanel() {
   219→   219→  const podeEditar = useAuth((s) => s.podeEditar());
   220→   220→  const queryClient = useQueryClient();
   221→   221→
   222→   222→  const q = useQuery<Despesa[]>({
   223→   223→    queryKey: ["despesas"],
   224→   224→    queryFn: () => apiFetch<Despesa[]>("/api/despesas"),
   225→   225→  });
   226→   226→
   227→   227→  const [novoNome, setNovoNome] = useState("");
   228→   228→  const [novoValor, setNovoValor] = useState("");
   229→   229→
   230→   230→  const total = useMemo(
   231→   231→    () => (q.data ?? []).reduce((acc, d) => acc + num(d.valor), 0),
   232→   232→    [q.data]
   233→   233→  );
   234→   234→
   235→   235→  const addMut = useMutation({
   236→   236→    mutationFn: (vars: { nome: string; valor: number }) =>
   237→   237→      apiFetch<Despesa>("/api/despesas", {
   238→   238→        method: "POST",
   239→   239→        body: JSON.stringify(vars),
   240→   240→      }),
   241→   241→    onSuccess: () => {
   242→   242→      queryClient.invalidateQueries({ queryKey: ["despesas"] });
   243→   243→      queryClient.invalidateQueries({ queryKey: ["resumo"] });
   244→   244→      setNovoNome("");
   245→   245→      setNovoValor("");
   246→   246→      toast.success("Despesa adicionada");
   247→   247→    },
   248→   248→    onError: (e: Error) => toast.error(e.message),
   249→   249→  });
   250→   250→
   251→   251→  const putMut = useMutation({
   252→   252→    mutationFn: (vars: { id: string; nome: string; valor: number }) =>
   253→   253→      apiFetch<Despesa>(`/api/despesas/${vars.id}`, {
   254→   254→        method: "PUT",
   255→   255→        body: JSON.stringify({ nome: vars.nome, valor: vars.valor }),
   256→   256→      }),
   257→   257→    onSuccess: () => {
   258→   258→      queryClient.invalidateQueries({ queryKey: ["despesas"] });
   259→   259→      queryClient.invalidateQueries({ queryKey: ["resumo"] });
   260→   260→      toast.success("Despesa atualizada");
   261→   261→    },
   262→   262→    onError: (e: Error) => toast.error(e.message),
   263→   263→  });
   264→   264→
   265→   265→  const delMut = useMutation({
   266→   266→    mutationFn: (id: string) =>
   267→   267→      apiFetch<void>(`/api/despesas/${id}`, { method: "DELETE" }),
   268→   268→    onSuccess: () => {
   269→   269→      queryClient.invalidateQueries({ queryKey: ["despesas"] });
   270→   270→      queryClient.invalidateQueries({ queryKey: ["resumo"] });
   271→   271→      toast.success("Despesa removida");
   272→   272→    },
   273→   273→    onError: (e: Error) => toast.error(e.message),
   274→   274→  });
   275→   275→
   276→   276→  function handleAdd() {
   277→   277→    const nome = novoNome.trim();
   278→   278→    const valor = num(novoValor);
   279→   279→    if (!nome) {
   280→   280→      toast.error("Informe o nome da despesa");
   281→   281→      return;
   282→   282→    }
   283→   283→    if (!(valor >= 0)) {
   284→   284→      toast.error("Informe um valor válido");
   285→   285→      return;
   286→   286→    }
   287→   287→    addMut.mutate({ nome, valor });
   288→   288→  }
   289→   289→
   290→   290→  return (
   291→   291→    <motion.div
   292→   292→      initial={{ opacity: 0, y: 4 }}
   293→   293→      animate={{ opacity: 1, y: 0 }}
   294→   294→      transition={{ duration: 0.2 }}
   295→   295→      className="space-y-4"
   296→   296→    >
   297→   297→      <Card className="bg-[var(--surface-app)] border-[var(--border-app)] py-0 shadow-none">
   298→   298→        <div className="px-4 py-3 border-b border-[var(--border-app-subtle)] flex items-center justify-between gap-2">
   299→   299→          <h3 className="text-sm font-semibold text-[var(--text-app)]">
   300→   300→            Despesas fixas mensais
   301→   301→          </h3>
   302→   302→          <span className="text-[11px] text-[var(--text-app-muted)]">
   303→   303→            {(q.data ?? []).length}{" "}
   304→   304→            {(q.data ?? []).length === 1 ? "item" : "itens"}
   305→   305→          </span>
   306→   306→        </div>
   307→   307→
   308→   308→        {/* Inline add form */}
   309→   309→        {podeEditar && (
   310→   310→          <div className="px-4 py-3 border-b border-[var(--border-app-subtle)] flex flex-col sm:flex-row sm:items-end gap-3">
   311→   311→            <div className="flex-1">
   312→   312→              <Label className="text-[11px] text-[var(--text-app-muted)] mb-1 block">
   313→   313→                Nome da despesa
   314→   314→              </Label>
   315→   315→              <Input
   316→   316→                value={novoNome}
   317→   317→                onChange={(e) => setNovoNome(e.target.value)}
   318→   318→                placeholder="Ex.: Aluguel, Energia, Internet…"
   319→   319→                className={inputCls}
   320→   320→                onKeyDown={(e) => {
   321→   321→                  if (e.key === "Enter") handleAdd();
   322→   322→                }}
   323→   323→              />
   324→   324→            </div>
   325→   325→            <div className="w-full sm:w-40">
   326→   326→              <Label className="text-[11px] text-[var(--text-app-muted)] mb-1 block">
   327→   327→                Valor médio (R$)
   328→   328→              </Label>
   329→   329→              <Input
   330→   330→                type="number"
   331→   331→                inputMode="decimal"
   332→   332→                step="0.01"
   333→   333→                min="0"
   334→   334→                value={novoValor}
   335→   335→                onChange={(e) => setNovoValor(e.target.value)}
   336→   336→                placeholder="0,00"
   337→   337→                className={`${inputCls} font-mono tabular-nums text-right`}
   338→   338→                onKeyDown={(e) => {
   339→   339→                  if (e.key === "Enter") handleAdd();
   340→   340→                }}
   341→   341→              />
   342→   342→            </div>
   343→   343→            <Button
   344→   344→              type="button"
   345→   345→              size="sm"
   346→   346→              onClick={handleAdd}
   347→   347→              disabled={addMut.isPending}
   348→   348→              className="bg-[var(--accent-app)] text-white hover:bg-[var(--accent-app-hover)] shrink-0"
   349→   349→            >
   350→   350→              <Plus size={14} />
   351→   351→              {addMut.isPending ? "Adicionando…" : "Adicionar"}
   352→   352→            </Button>
   353→   353→          </div>
   354→   354→        )}
   355→   355→
   356→   356→        {/* Table / states */}
   357→   357→        {q.isLoading ? (
   358→   358→          <TableSkeleton cols={3} />
   359→   359→        ) : q.isError ? (
   360→   360→          <ErrorState
   361→   361→            message={q.error?.message ?? "Falha ao carregar despesas"}
   362→   362→            onRetry={() => q.refetch()}
   363→   363→          />
   364→   364→        ) : (q.data ?? []).length === 0 ? (
   365→   365→          <EmptyState
   366→   366→            icon={<Wallet size={22} strokeWidth={1.5} />}
   367→   367→            title="Nenhuma despesa cadastrada"
   368→   368→            hint={
   369→   369→              podeEditar
   370→   370→                ? "Use o formulário acima para adicionar aluguel, energia, salários, etc. Esses custos entram no cálculo do custo fixo por minuto."
   371→   371→                : "Aguarde o gestor cadastrar as despesas fixas da clínica."
   372→   372→            }
   373→   373→          />
   374→   374→        ) : (
   375→   375→          <div className="max-h-96 overflow-y-auto scroll-thin">
   376→   376→            <Table className="text-sm">
   377→   377→              <TableHeader>
   378→   378→                <TableRow className="border-[var(--border-app)] hover:bg-transparent">
   379→   379→                  <TableHead className="text-[var(--text-app-muted)] text-[11px] uppercase tracking-wide font-normal px-4 py-2">
   380→   380→                    Despesa
   381→   381→                  </TableHead>
   382→   382→                  <TableHead className="text-right text-[var(--text-app-muted)] text-[11px] uppercase tracking-wide font-normal px-4 py-2 w-40">
   383→   383→                    Valor médio mensal
   384→   384→                  </TableHead>
   385→   385→                  <TableHead className="w-12 px-2" />
   386→   386→                </TableRow>
   387→   387→              </TableHeader>
   388→   388→              <TableBody>
   389→   389→                {(q.data ?? []).map((d) => (
   390→   390→                  <DespesaRow
   391→   391→                    key={d.id}
   392→   392→                    despesa={d}
   393→   393→                    podeEditar={podeEditar}
   394→   394→                    onSalvar={(nome, valor) =>
   395→   395→                      putMut.mutate({ id: d.id, nome, valor })

   397→   397→                    onRemover={() => delMut.mutate(d.id)}

   401→   401→              </TableBody>

   405→   405→                    Total
   406→   406→                  </TableCell>
   407→   407→                  <TableCell className="px-4 py-2 text-right font-mono tabular-nums text-[var(--text-app)]">
   408→   408→                    {brl(total)}

   410→   410→                  <TableCell />
   411→   411→                </TableRow>

   413→   413→            </Table>
   414→   414→          </div>
   415→   415→        )}
   416→   416→      </Card>
   417→   417→
   418→   418→      <p className="text-[11px] text-[var(--text-app-muted)] px-1">
   419→   419→        As despesas fixas compõem o custo fixo mensal da clínica e entram no
   420→   420→        rateio por minuto para a precificação de procedimentos.
   421→   421→      </p>
   422→   422→    </motion.div>
   423→   423→  );
   424→   424→}
   425→   425→
   426→   426→function DespesaRow({
   427→   427→  despesa,
   428→   428→  podeEditar,
   429→   429→  onSalvar,
   430→   430→  onRemover,
   431→   431→  salvando,
   432→   432→}: {
   433→   433→  despesa: Despesa;
   434→   434→  podeEditar: boolean;
   435→   435→  onSalvar: (nome: string, valor: number) => void;
   436→   436→  onRemover: () => void;
   437→   437→  salvando: boolean;
   438→   438→}) {
   439→   439→  // Uncontrolled inputs with defaultValue + key={d.id} (from parent).
   440→   440→  // Initial value comes from the server on first mount; subsequent refetches
   441→   441→  // with the same id keep the user's local edits (no re-init).
   442→   442→  function commitNome(novoNome: string) {
   443→   443→    const n = novoNome.trim();
   444→   444→    if (!n) {
   445→   445→      toast.error("Nome não pode ser vazio");
   446→   446→      return;
   447→   447→    }
   448→   448→    if (n !== despesa.nome) {
   449→   449→      onSalvar(n, num(despesa.valor));
   450→   450→    }
   451→   451→  }
   452→   452→
   453→   453→  function commitValor(novoValor: string) {
   454→   454→    const v = num(novoValor);
   455→   455→    if (!(v >= 0)) {
   456→   456→      toast.error("Valor inválido");
   457→   457→      return;
   458→   458→    }
   459→   459→    if (v !== num(despesa.valor)) {
   460→   460→      onSalvar(despesa.nome, v);
   461→   461→    }
   462→   462→  }
   463→   463→
   464→   464→  return (
   465→   465→    <TableRow className="border-t border-[var(--border-app-subtle)] hover:bg-[var(--bg-app)]/40">
   466→   466→      <TableCell className="px-4 py-1.5">
   467→   467→        <Input
   468→   468→          defaultValue={despesa.nome}
   469→   469→          onBlur={(e) => commitNome(e.currentTarget.value)}
   470→   470→          disabled={!podeEditar || salvando}
   471→   471→          className={`${inputCls} ${cellInputCls} ${
   472→   472→            podeEditar ? "" : "opacity-70"
   473→   473→          }`}
   474→   474→        />
   475→   475→      </TableCell>
   476→   476→      <TableCell className="px-4 py-1.5">
   477→   477→        <Input
   478→   478→          type="number"
   479→   479→          inputMode="decimal"

  1147→  1147→  onSalvar,
